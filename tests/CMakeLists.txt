# Run with: cmake -S tests -B /tmp/microros-toolchain-tests
cmake_minimum_required(VERSION 3.16)
project(microros_toolchain_tests NONE)

include("${CMAKE_CURRENT_LIST_DIR}/../cmake/microros_flags.cmake")
set(test_dir "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
file(MAKE_DIRECTORY "${test_dir}/with spaces")

function(idf_build_get_property output property)
    set(${output} "${test_${property}}" PARENT_SCOPE)
endfunction()

function(assert_flag variable expected)
    separate_arguments(arguments UNIX_COMMAND "${${variable}}")
    if(NOT "${expected}" IN_LIST arguments)
        message(FATAL_ERROR "Missing '${expected}' in ${variable}: ${arguments}")
    endif()
endfunction()

# IDF 5-style strings plus project-level options, including PSRAM workarounds.
set(CMAKE_C_FLAGS "-march=rv32imc_zicsr_zifencei -mabi=ilp32")
set(CMAKE_CXX_FLAGS "-march=rv32imac_zicsr_zifencei -mabi=ilp32")
set(test_COMPILE_OPTIONS "-mfix-esp32-psram-cache-issue;-Og;-g;-fstack-protector-strong;-Wall;-Werror;-Wp,-DTEST_FORWARD=1")
set(test_C_COMPILE_OPTIONS "-std=gnu17;-Wstrict-prototypes")
set(test_CXX_COMPILE_OPTIONS "-std=gnu++26;-fno-exceptions;-fno-rtti")
set(test_COMPILE_DEFINITIONS "ESP_PLATFORM;TEXT=\"hello world\"")
microros_get_flags(C c_flags)
microros_get_flags(CXX cxx_flags)
assert_flag(c_flags "-march=rv32imc_zicsr_zifencei")
assert_flag(cxx_flags "-march=rv32imac_zicsr_zifencei")
assert_flag(c_flags "-mfix-esp32-psram-cache-issue")
assert_flag(c_flags "-Og")
assert_flag(c_flags "-fstack-protector-strong")
assert_flag(c_flags "-Wp,-DTEST_FORWARD=1")
assert_flag(c_flags "-DTEXT=\"hello world\"")
assert_flag(cxx_flags "-fno-exceptions")
if(c_flags MATCHES "-Wall|-Werror|-std=|-fno-exceptions" OR cxx_flags MATCHES "-std=|-Wstrict-prototypes")
    message(FATAL_ERROR "Warnings, language standards or language-specific flags leaked")
endif()

# IDF 6 response files, mixed with plain flags and nested response files.
file(WRITE "${test_dir}/with spaces/nested" "-march=rv32imc_zicsr_zifencei\n-mabi=ilp32\n")
file(WRITE "${test_dir}/flags" "@\"${test_dir}/with spaces/nested\"\n\"-specs=${test_dir}/with spaces/picolibc.specs\"\n")
set(CMAKE_C_FLAGS "@\"${test_dir}/flags\" -fno-builtin-memcpy")
microros_get_flags(C c_flags)
assert_flag(c_flags "-march=rv32imc_zicsr_zifencei")
assert_flag(c_flags "-specs=${test_dir}/with spaces/picolibc.specs")
assert_flag(c_flags "-fno-builtin-memcpy")

# Changing response contents at the same path must change the exported flags.
file(WRITE "${test_dir}/with spaces/nested" "-march=rv32imac_zicsr_zifencei\n-mabi=ilp32\n")
microros_get_flags(C updated_flags)
assert_flag(updated_flags "-march=rv32imac_zicsr_zifencei")
if(updated_flags STREQUAL c_flags)
    message(FATAL_ERROR "Response file content change was lost")
endif()

# Repeated toolchain loads must replace stale cache entries, not accumulate flags.
set(MICROROS_C_FLAGS "${c_flags}")
set(MICROROS_CXX_FLAGS "${cxx_flags}")
set(BUILD_CONFIG_DIR "${test_dir}")
set(IDF_TARGET_ARCH riscv)
set(IDF_TARGET esp32c3)
configure_file("${CMAKE_CURRENT_LIST_DIR}/../esp32_toolchain.cmake.in" "${test_dir}/toolchain.cmake" @ONLY)
set(CMAKE_C_FLAGS "-march=wrong" CACHE STRING "" FORCE)
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG" CACHE STRING "" FORCE)
set(MICROROS_TOOLCHAIN_FLAGS_APPLIED TRUE CACHE INTERNAL "Old toolchain guard")
include("${test_dir}/toolchain.cmake")
set(first_flags "${CMAKE_C_FLAGS}")
include("${test_dir}/toolchain.cmake")
if(NOT first_flags STREQUAL CMAKE_C_FLAGS OR CMAKE_C_FLAGS MATCHES "march=wrong")
    message(FATAL_ERROR "Toolchain retained stale flags or duplicated them")
endif()
set(MICROROS_C_FLAGS "${updated_flags}")
configure_file("${CMAKE_CURRENT_LIST_DIR}/../esp32_toolchain.cmake.in" "${test_dir}/toolchain.cmake" @ONLY)
include("${test_dir}/toolchain.cmake")
assert_flag(CMAKE_C_FLAGS "-march=rv32imac_zicsr_zifencei")
if(CMAKE_C_FLAGS_RELEASE OR CMAKE_CXX_FLAGS_RELEASE)
    message(FATAL_ERROR "Release defaults can override the IDF optimization/assertion flags")
endif()

message(STATUS "micro-ROS toolchain flag tests passed")
